  size(240, 120);
  shapeMode(CENTER);

void draw() {
  background(0);
  float diameter = map(mouseX, 0, width, 10, 800);
  shape(network, 120,  60,  diameter, diameter);
}

07_011
		
PShape dino;
void setup() {
  size(480, 120);
  dino = createShape();
  dino.beginShape();
  dino.fill(153, 176, 180);
  dino.vertex(50, 120);
  dino.vertex(100, 90);
  dino.vertex(110, 60);
  dino.vertex(80, 20);
  dino.vertex(210, 60);
  dino.vertex(160, 80);
  dino.vertex(200, 90);
  dino.vertex(140, 100);
  dino.vertex(130, 120);
  dino.endShape();
}
void draw() {
  background(204);
  translate(mouseX - 120, 0);
  shape(dino, 0, 0);
}
  • PShapeクラス
  • createShape()
  • dinoはPShapeクラスのインスタンス
  • dinoはオブジェクト
  • beginShape(), fill(), vertex(), endShape()
  • PShapeクラスのメソッド

まとめ

  • ProcessingはJPEG, GIF, PNG, SVGなどの画像ファイルを扱える
  • ラスター画像ファイルはPImageクラス，
  • ベクター画像ファイルはPShapeクラスを用いて利用できる．
  • 文字列はStringクラスを用いて利用できる．
