落ちる円

	        
float y=0;
void setup() {
  size(400, 300);
}
void draw() {
  background(204);
  ellipse(width/2, y, 40, 40);
  y++;
}

}
void draw() {
  background(204);

  }
}

ASCIIコードで表すと'A'=0d65 から 'Z'=0d91

ランダムな位置で落ちる文字

	        
int c;
int fontsize=24;                  
float x;                  
float y=0;
void setup() {
  size(400, 300);
  textSize(fontsize);
  fill(0);
  x=random(fontsize, width-fontsize);
  c=int(random(65, 91));
}
void draw() {
  background(204);
  text(char(c), x, y); 
  y++;
  if (y>height) {
    x=random(24, width-24);
    y=0;
    c=int(random(65, 91));
  }
}

$24\leqq x \leqq \textrm{width}-24$

keyPressed()

	        
int c;
int fontsize=24;                  
float x;                  
float y=0;
void setup() {
  size(400, 300);
  textSize(fontsize);
  fill(0);
  c=int(random(65, 91));
  x=random(fontsize, width-fontsize);
}
void draw() {
  background(204);
  text(char(c), x, y); 
  y++;
  if (y>height) {
    x=random(fontsize, width-fontsize);
    y=0;
    c=int(random(65, 91));
  }
}
void keyPressed() {
  if (key==char(c)||key==char(c+32)) {
    x=random(fontsize, width-fontsize);
    y=0;
    c=int(random(65, 91));
  }
}

$24\leqq x \leqq \textrm{width}-24$
