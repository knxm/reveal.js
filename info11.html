オブジェクトの作り方

  • オブジェクトの設計図を作る
  • 設計図 → クラス
  • 1つの設計図（クラス）から多数のオブジェクトを生成
  • オブジェクトの1つ1つは異なる値をフィールドに保存
同じ種類のスマートフォン
  • 色，メモリ容量，機種番号等異なる．
  • しかし，元の設計図は同じ
  • クラス（設計図）→オブジェクト（もの）
  • オブジェクト=クラスのインスタンス

タブ機能

  • PDEにはタブ機能が存在
  • Processingの基本は同一名のフォルダとファイルのペア
  • コードのファイル名拡張子は .pde

  • 設定変更で表示可能
  • フォルダ内に複数のファイル→タブに表示
  • 共通利用するクラスを別ファイルで保存
  • クラス名は大文字開始
  • 複数コードからなる開発が可能
  • クラスの開発を分担可能

Star クラス

$(x, y)$座標と大きさ$d$を指定して星を描く
  1. クラス Star の作成
  2. フィールド $x, y, d$ の追加
  3. コンストラクタの作成
  4. フィールドの代入
  5. メソッド display() の追加
  6. 先週作成した関数 drawStar() を利用

クラス Star の作成

	        
// クラスの作成
class Star {
// フィールドの追加
  float x;
  float y;
  float d;
// コンストラクタの作成
  Star(float tmpX, float tmpY, float tmpD) {
// フィールドの代入
    x = tmpX;
    y = tmpY;
    d = tmpD;
  }
// メソッドの追加
  void display() {
    noStroke();
    pushMatrix();
    translate(x, y);
    beginShape();
    for (int i=0; i < 5; i++) {
      float t = - HALF_PI + 2*i*TWO_PI/5;
      vertex((d/2.0)*cos(t), (d/2.0)*sin(t));
    }
    endShape(CLOSE);
    popMatrix();
  }
}

廻る星

	        
// クラスの作成
class Star {
// フィールドの追加
  float x;
  float y;
  float d;
  float a = 0; // 廻すための角度 a を追加
// コンストラクタの作成
  Star(float tmpX, float tmpY, float tmpD) {
// フィールドの代入
    x = tmpX;
    y = tmpY;
    d = tmpD;
  }
// メソッドの追加
  void display() {
    noStroke();
    pushMatrix();
    translate(x, y);
    rotate(a); // 座標軸を時計回りに回転
    beginShape();
    for (int i=0; i < 5; i++) {
      float t = - HALF_PI + 2*i*TWO_PI/5;
      vertex((d/2.0)*cos(t), (d/2.0)*sin(t));
    }
    endShape(CLOSE);
    popMatrix();
    a+=0.1; // 描く度に0.1ラジアン増加
  }
}

跳ね廻る星

	        
Star star1;
void setup() {
  size(300, 300);
  star1 = new Star(200, 100, 20);
}
void draw() {
  background(0);
  fill(255, 255, 0);
  star1.move(); // 星を動かす 
  star1.checkEdge(); // 壁との反射確認
  star1.display(); // 星を表示
}
// クラスの作成
class Star {
  float x;
  float y;
  float d;
  float a = 0;
  float r = d/2.0;
  float vx = random(1, 4); // x方向の速度 
  float vy = random(1, 4); // y方向の速度
  Star(float tmpX, float tmpY, float tmpD) {
    x = tmpX;
    y = tmpY;
    d = tmpD;
  }
  void display() {
    noStroke();
    pushMatrix();
    translate(x, y);
    rotate(a);
    beginShape();
    for (int i=0; i < 5; i++) {
      float t = - HALF_PI + 2*i*TWO_PI/5;
      vertex((d/2.0)*cos(t), (d/2.0)*sin(t));
    }
      endShape(CLOSE);
      popMatrix();
      a+=0.1;
    }
    // x方向，y方向の速度                  
    void move() {
      x += vx;
      y += vy;
    }
    void checkEdge() {
      // 左右の壁                      
      if (x < r || x > width - r) {
        vx *= -1; 
      }
      // 上下の壁
      if (y < r || y > height - r) {
        vy *= -1; 
      }
    }                      
}
